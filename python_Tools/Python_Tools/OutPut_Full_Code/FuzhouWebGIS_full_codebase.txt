def get_shortest_path(start_coords, end_coords, mode: str = "car") -> dict:
    """
    使用 PostGIS + pgRouting 计算最短路径。

    参数:
    - start_coords: (lng, lat)，WGS84 (EPSG:4326)
    - end_coords: (lng, lat)，WGS84 (EPSG:4326)
    - mode: "car" 或 "walk"
    """
    if mode not in ("car", "walk"):
        return {"success": False, "message": "mode 仅支持 'car' 或 'walk'"}

    if not isinstance(start_coords, (list, tuple)) or not isinstance(end_coords, (list, tuple)):
        return {"success": False, "message": "start_coords 和 end_coords 必须是 (lng, lat)"}

    if len(start_coords) != 2 or len(end_coords) != 2:
        return {"success": False, "message": "坐标格式错误，应为 (lng, lat)"}

    try:
        start_lng, start_lat = float(start_coords[0]), float(start_coords[1])
        end_lng, end_lat = float(end_coords[0]), float(end_coords[1])
    except (TypeError, ValueError):
        return {"success": False, "message": "坐标必须是数字"}

    conn = None
    cur = None
    try:
        conn = get_db_connection()
        cur = conn.cursor()

        nearest_any_sql = """
        SELECT
            id,
            ST_Distance(
                geom,
                ST_Transform(ST_SetSRID(ST_MakePoint(%s, %s), 4326), 3857)
            ) AS dist_m
        FROM fuzhou_roads_vertices
        ORDER BY geom <-> ST_Transform(ST_SetSRID(ST_MakePoint(%s, %s), 4326), 3857)
        LIMIT 1;
        """

        nearest_edge_cost_sql = """
        SELECT cost_car_s
        FROM fuzhou_roads_clean
        ORDER BY geom <-> ST_Transform(ST_SetSRID(ST_MakePoint(%s, %s), 4326), 3857)
        LIMIT 1;
        """

        nearest_drivable_sql = """
        WITH drivable_vertices AS (
            SELECT DISTINCT v.id, v.geom
            FROM fuzhou_roads_vertices v
            JOIN fuzhou_roads_clean r ON v.id = r.source OR v.id = r.target
            WHERE r.cost_car_s > 0
        )
        SELECT
            id,
            ST_Distance(
                geom,
                ST_Transform(ST_SetSRID(ST_MakePoint(%s, %s), 4326), 3857)
            ) AS dist_m
        FROM drivable_vertices
        ORDER BY geom <-> ST_Transform(ST_SetSRID(ST_MakePoint(%s, %s), 4326), 3857)
        LIMIT 1;
        """

        def nearest_vertex(lng: float, lat: float, drivable_only: bool = False) -> Optional[dict]:
            sql = nearest_drivable_sql if drivable_only else nearest_any_sql
            cur.execute(sql, (lng, lat, lng, lat))
            return cur.fetchone()

        def build_path(start_vid: int, end_vid: int, mode_name: str) -> Optional[dict]:
            cost_field = "cost_car_s" if mode_name == "car" else "cost_walk_s"
            edge_sql = (
                f"SELECT id, source, target, {cost_field} * traffic_factor AS cost "
                f"FROM fuzhou_roads_clean WHERE {cost_field} >= 0"
            )

            route_sql = f"""
            WITH route AS (
                SELECT *
                FROM pgr_dijkstra(
                    '{edge_sql}',
                    %s,
                    %s,
                    FALSE
                )
            )
            SELECT
                rt.seq,
                rt.edge,
                rt.cost,
                r.traffic_factor,
                ST_Length(r.geom) AS edge_distance_m,
                ST_AsGeoJSON(ST_Transform(r.geom, 4326)) AS edge_geojson
            FROM route rt
            JOIN fuzhou_roads_clean r ON rt.edge = r.id
            WHERE rt.edge <> -1
            ORDER BY rt.seq;
            """
            cur.execute(route_sql, (start_vid, end_vid))
            rows = cur.fetchall()
            if not rows:
                return None

            features = []
            total_cost_s = 0.0
            total_distance_m = 0.0

            for row in rows:
                edge_cost = float(row.get("cost") or 0.0)
                edge_distance = float(row.get("edge_distance_m") or 0.0)
                traffic_factor = float(row.get("traffic_factor") or 1.0)
                edge_geojson = row.get("edge_geojson")
                if not edge_geojson:
                    continue

                total_cost_s += edge_cost
                total_distance_m += edge_distance

                features.append(
                    {
                        "type": "Feature",
                        "properties": {
                            "mode": mode_name,
                            "segment_type": mode_name,
                            "cost_seconds": edge_cost,
                            "distance_m": edge_distance,
                            "traffic_factor": traffic_factor,
                        },
                        "geometry": json.loads(edge_geojson),
                    }
                )

            if not features:
                return None

            return {
                "geojson": {
                    "type": "FeatureCollection",
                    "features": features,
                },
                "total_cost_s": total_cost_s,
                "distance_m": total_distance_m,
                "edge_count": len(features),
            }

        if mode == "walk":
            start_vertex = nearest_vertex(start_lng, start_lat, drivable_only=False)
            end_vertex = nearest_vertex(end_lng, end_lat, drivable_only=False)
            if not start_vertex or not end_vertex:
                return {"success": False, "message": "无法匹配到最近步行路网节点"}

            start_vid = int(start_vertex["id"])
            end_vid = int(end_vertex["id"])
            walk_path = build_path(start_vid, end_vid, "walk")
            if not walk_path:
                return {
                    "success": False,
                    "message": "未找到可用步行路径，请尝试更换起终点",
                    "start_vertex": start_vid,
                    "end_vertex": end_vid,
                }

            return {
                "success": True,
                "mode": mode,
                "start": {"lng": start_lng, "lat": start_lat, "vertex_id": start_vid},
                "end": {"lng": end_lng, "lat": end_lat, "vertex_id": end_vid},
                "total_cost_s": walk_path["total_cost_s"],
                "total_duration_s": walk_path["total_cost_s"],
                "distance_m": walk_path["distance_m"],
                "duration_mins": round(walk_path["total_cost_s"] / 60.0, 2),
                "eta_minutes": round(walk_path["total_cost_s"] / 60.0, 2),
                "edge_count": walk_path["edge_count"],
                "geojson": walk_path["geojson"],
            }

        # mode == "car": 最后一公里多模态
        cur.execute(nearest_edge_cost_sql, (end_lng, end_lat))
        nearest_edge_row = cur.fetchone()
        if nearest_edge_row and nearest_edge_row.get("cost_car_s") == -1:
            return {
                "success": False,
                "message": "目的地附近最近路段不允许驾车通行（cost_car_s = -1），请改用步行或调整终点",
            }

        start_car_vertex = nearest_vertex(start_lng, start_lat, drivable_only=True)
        end_car_vertex = nearest_vertex(end_lng, end_lat, drivable_only=True)
        end_walk_vertex = nearest_vertex(end_lng, end_lat, drivable_only=False)

        if not start_car_vertex or not end_car_vertex or not end_walk_vertex:
            return {"success": False, "message": "无法匹配到可驾车或步行路网节点"}

        start_car_vid = int(start_car_vertex["id"])
        end_car_vid = int(end_car_vertex["id"])
        end_walk_vid = int(end_walk_vertex["id"])
        last_mile_dist_m = float(end_car_vertex.get("dist_m") or 0.0)

        car_path = build_path(start_car_vid, end_car_vid, "car")
        if not car_path:
            return {
                "success": False,
                "message": "未找到可用驾车路径，请尝试更换起终点",
                "start_vertex": start_car_vid,
                "end_vertex": end_car_vid,
            }

        walk_path = None
        if last_mile_dist_m <= LAST_MILE_THRESHOLD_M and end_car_vid != end_walk_vid:
            walk_path = build_path(end_car_vid, end_walk_vid, "walk")

        features = list(car_path["geojson"].get("features", []))

        total_cost_s = car_path["total_cost_s"]
        total_distance_m = car_path["distance_m"]
        total_edges = car_path["edge_count"]

        if walk_path:
            walk_features = walk_path["geojson"].get("features", [])
            for feature in walk_features:
                props = feature.setdefault("properties", {})
                props["mode"] = "car"
                props["segment_type"] = "walk"
            features.extend(walk_features)
            total_cost_s += walk_path["total_cost_s"]
            total_distance_m += walk_path["distance_m"]
            total_edges += walk_path["edge_count"]

        return {
            "success": True,
            "mode": mode,
            "start": {"lng": start_lng, "lat": start_lat, "vertex_id": start_car_vid},
            "end": {"lng": end_lng, "lat": end_lat, "vertex_id": end_walk_vid},
            "last_mile_threshold_m": LAST_MILE_THRESHOLD_M,
            "last_mile_distance_m": round(last_mile_dist_m, 2),
            "contains_last_mile_walk": bool(walk_path),
            "total_cost_s": total_cost_s,
            "total_duration_s": total_cost_s,
            "distance_m": total_distance_m,
            "duration_mins": round(total_cost_s / 60.0, 2),
            "eta_minutes": round(total_cost_s / 60.0, 2),
            "edge_count": total_edges,
            "geojson": {
                "type": "FeatureCollection",
                "features": features,
            },
        }
    except psycopg2.Error as exc:
        return {"success": False, "message": f"路径计算失败: {exc}"}
    finally:
        if cur:
            cur.close()
        if conn:
            conn.close()


def parse_polyline(polyline: str) -> List[List[float]]:
    coords: List[List[float]] = []
    if not polyline:
        return coords

    for pair in polyline.split(";"):
        parts = pair.split(",")
        if len(parts) != 2:
            continue
        try:
            lng = float(parts[0])
            lat = float(parts[1])
        except ValueError:
            continue
        coords.append([lng, lat])
    return coords


def call_amap(api: str, params: Dict[str, str]) -> dict:
    if not AMAP_KEY:
        raise HTTPException(
            status_code=500,
            detail="未配置高德地图 Key，请设置环境变量 AMAP_KEY 后重试。",
        )

    query = {"key": AMAP_KEY, **params}
    url = f"https://restapi.amap.com{api}?{urlencode(query)}"

    try:
        with urlopen(url, timeout=12) as resp:
            body = resp.read().decode("utf-8")
            payload = json.loads(body)
    except Exception as exc:
        raise HTTPException(status_code=502, detail=f"路径服务调用失败: {exc}")

    if payload.get("status") != "1":
        raise HTTPException(
            status_code=502,
            detail=f"路径服务返回错误: {payload.get('info', '未知错误')}",
        )

    return payload


def call_osrm(profile: str, origin_lng: float, origin_lat: float, dest_lng: float, dest_lat: float) -> dict:
    if profile not in ("driving", "walking"):
        raise HTTPException(status_code=400, detail="不支持的路线模式")

    query = urlencode(
        {
            "overview": "full",
            "alternatives": "true",
            "geometries": "geojson",
            "steps": "true",
        }
    )
    url = (
        "https://router.project-osrm.org/route/v1/"
        f"{profile}/{origin_lng},{origin_lat};{dest_lng},{dest_lat}?{query}"
    )

    try:
        with urlopen(url, timeout=12) as resp:
            body = resp.read().decode("utf-8")
            payload = json.loads(body)
    except Exception as exc:
        raise HTTPException(status_code=502, detail=f"OSRM 路径服务调用失败: {exc}")

    if payload.get("code") != "Ok":
        raise HTTPException(
            status_code=502,
            detail=f"OSRM 路径服务返回错误: {payload.get('message', '未知错误')}",
        )

    return payload


def classify_traffic_by_speed(speed_kmh: float) -> str:
    if speed_kmh >= 35:
        return "畅通"
    if speed_kmh >= 20:
        return "缓行"
    if speed_kmh >= 10:
        return "拥堵"
    return "严重拥堵"

@app.get("/api/search")
async def search_poi(
    keyword: str = Query(..., min_length=1, description="按地点名称模糊匹配"),
    category: Optional[str] = Query(None, description="按类型过滤"),
    limit: int = Query(20, ge=1, le=100),
):
    conn = None
    cur = None
    try:
        conn = get_db_connection()
        cur = conn.cursor()

        kw = keyword.strip()
        if not kw:
            return {"data": []}

        if category:
            sql = """
            SELECT name, category, ST_AsGeoJSON(ST_Transform(geom, 4326)) AS geometry
            FROM fuzhou_poi_clean
            WHERE category = %s
              AND name ILIKE %s
            ORDER BY CASE WHEN name ILIKE %s THEN 0 ELSE 1 END, name
            LIMIT %s;
            """
            cur.execute(sql, (category, f"%{kw}%", f"{kw}%", limit))
        else:
            sql = """
            SELECT name, category, ST_AsGeoJSON(ST_Transform(geom, 4326)) AS geometry
            FROM fuzhou_poi_clean
            WHERE name ILIKE %s
            ORDER BY CASE WHEN name ILIKE %s THEN 0 ELSE 1 END, name
            LIMIT %s;
            """
            cur.execute(sql, (f"%{kw}%", f"{kw}%", limit))

        return {"data": cur.fetchall()}
    except psycopg2.Error as exc:
        raise HTTPException(status_code=500, detail=f"地点搜索失败: {exc}")
    finally:
        if cur:
            cur.close()
        if conn:
            conn.close()


@app.get("/api/nearby")
async def nearby_poi(
    lng: float = Query(..., description="当前位置经度"),
    lat: float = Query(..., description="当前位置纬度"),
    radius: int = Query(2000, ge=100, le=20000, description="搜索半径，单位米"),
    category: Optional[str] = Query(None, description="按类型过滤"),
    limit: int = Query(100, ge=1, le=500),
):
    conn = None
    cur = None
    try:
        conn = get_db_connection()
        cur = conn.cursor()

        sql = """
        SELECT
            name,
            category,
            ST_AsGeoJSON(ST_Transform(geom, 4326)) AS geometry,
            ST_Distance(
                ST_Transform(geom, 4326)::geography,
                ST_SetSRID(ST_MakePoint(%s, %s), 4326)::geography
            ) AS distance_m
        FROM fuzhou_poi_clean
        WHERE (%s IS NULL OR category = %s)
          AND ST_DWithin(
                ST_Transform(geom, 4326)::geography,
                ST_SetSRID(ST_MakePoint(%s, %s), 4326)::geography,
                %s
          )
        ORDER BY distance_m ASC
        LIMIT %s;
        """
        cur.execute(sql, (lng, lat, category, category, lng, lat, radius, limit))

        rows = cur.fetchall()
        for row in rows:
            row["distance_m"] = round(float(row.get("distance_m") or 0.0), 1)

        return {
            "data": rows,
            "radius_m": radius,
            "center": {"lng": lng, "lat": lat},
        }
    except psycopg2.Error as exc:
        raise HTTPException(status_code=500, detail=f"附近地点查询失败: {exc}")
    finally:
        if cur:
            cur.close()
        if conn:
            conn.close()

@app.post("/api/analyze")
async def analyze(payload: dict = Body(...)):
    geojson = payload["geojson"]
    search_type = payload["type"]
    categories: List[str] = payload.get("categories", [])

    if not categories:
        return {"data": []}

    conn = None
    cur = None
    try:
        conn = get_db_connection()
        cur = conn.cursor()

        geom_json = json.dumps(geojson["geometry"])

        if search_type == "polygon":
            sql = """
            SELECT name, category, ST_AsGeoJSON(ST_Transform(geom, 4326)) as geometry
            FROM fuzhou_poi_clean
            WHERE category = ANY(%s)
              AND ST_Intersects(geom, ST_Transform(ST_GeomFromGeoJSON(%s), 3857))
            LIMIT 20;
            """
            cur.execute(sql, (categories, geom_json))
        else:
            sql = """
            SELECT name, category, ST_AsGeoJSON(ST_Transform(geom, 4326)) as geometry
            FROM fuzhou_poi_clean
            WHERE category = ANY(%s)
            ORDER BY geom <-> ST_Transform(ST_GeomFromGeoJSON(%s), 3857)
            LIMIT 10;
            """
            cur.execute(sql, (categories, geom_json))

        results = cur.fetchall()
        return {"data": results}
    except psycopg2.Error as exc:
        raise HTTPException(status_code=500, detail=f"分析查询失败: {exc}")
    finally:
        if cur:
            cur.close()
        if conn:
            conn.close()


@app.get("/api/route")
async def plan_route(
    origin_lng: float = Query(...),
    origin_lat: float = Query(...),
    dest_lng: float = Query(...),
    dest_lat: float = Query(...),
    mode: str = Query("driving", pattern="^(driving|walking)$"),
):
    origin = f"{origin_lng},{origin_lat}"
    destination = f"{dest_lng},{dest_lat}"

    normalized_paths = []

    if AMAP_KEY:
        if mode == "driving":
            payload = call_amap(
                "/v3/direction/driving",
                {
                    "origin": origin,
                    "destination": destination,
                    "extensions": "all",
                    "strategy": "0",
                    "output": "json",
                },
            )
        else:
            payload = call_amap(
                "/v3/direction/walking",
                {
                    "origin": origin,
                    "destination": destination,
                    "output": "json",
                },
            )

        paths = payload.get("route", {}).get("paths", [])
        if not paths:
            raise HTTPException(status_code=404, detail="未查询到可用路线")

        for path in paths:
            distance_m = float(path.get("distance") or 0.0)
            duration_s = float(path.get("duration") or 0.0)
            steps = path.get("steps") or []

            segments = []
            traffic_counter: Dict[str, int] = {}

            if mode == "driving":
                for step in steps:
                    tmcs = step.get("tmcs") or []
                    if tmcs:
                        for tmc in tmcs:
                            status = tmc.get("status") or "未知"
                            traffic_counter[status] = traffic_counter.get(status, 0) + 1
                            coords = parse_polyline(tmc.get("polyline") or "")
                            if coords:
                                segments.append(
                                    {
                                        "status": status,
                                        "coords": coords,
                                    }
                                )
                    else:
                        coords = parse_polyline(step.get("polyline") or "")
                        if coords:
                            status = "未知"
                            traffic_counter[status] = traffic_counter.get(status, 0) + 1
                            segments.append(
                                {
                                    "status": status,
                                    "coords": coords,
                                }
                            )
            else:
                for step in steps:
                    coords = parse_polyline(step.get("polyline") or "")
                    if coords:
                        segments.append(
                            {
                                "status": "步行",
                                "coords": coords,
                            }
                        )

            normalized_paths.append(
                {
                    "distance_m": round(distance_m, 1),
                    "duration_s": round(duration_s, 1),
                    "strategy": path.get("strategy", ""),
                    "traffic": traffic_counter,
                    "segments": segments,
                }
            )
    else:
        profile = "driving" if mode == "driving" else "walking"
        payload = call_osrm(profile, origin_lng, origin_lat, dest_lng, dest_lat)
        routes = payload.get("routes") or []
        if not routes:
            raise HTTPException(status_code=404, detail="未查询到可用路线")

        for route in routes:
            coords = (route.get("geometry") or {}).get("coordinates") or []
            distance_m = float(route.get("distance") or 0.0)
            duration_s = float(route.get("duration") or 0.0)

            if mode == "driving":
                speed_kmh = (distance_m / duration_s) * 3.6 if duration_s > 0 else 0.0
                status = classify_traffic_by_speed(speed_kmh)
                traffic_counter = {status: 1}
            else:
                status = "步行"
                traffic_counter = {}

            normalized_paths.append(
                {
                    "distance_m": round(distance_m, 1),
                    "duration_s": round(duration_s, 1),
                    "strategy": "OSRM",
                    "traffic": traffic_counter,
                    "segments": [
                        {
                            "status": status,
                            "coords": coords,
                        }
                    ],
                }
            )

    return {
        "mode": mode,
        "origin": {"lng": origin_lng, "lat": origin_lat},
        "destination": {"lng": dest_lng, "lat": dest_lat},
        "provider": "amap" if AMAP_KEY else "osrm",
        "paths": normalized_paths,
    }


@app.get("/api/route_shortest")
async def route_shortest(
    origin_lng: float = Query(..., description="起点经度，4326"),
    origin_lat: float = Query(..., description="起点纬度，4326"),
    dest_lng: float = Query(..., description="终点经度，4326"),
    dest_lat: float = Query(..., description="终点纬度，4326"),
    mode: str = Query("car", pattern="^(car|walk)$"),
):
    result = get_shortest_path((origin_lng, origin_lat), (dest_lng, dest_lat), mode)
    if not result.get("success"):
        message = result.get("message", "最短路径计算失败")
        status_code = 404 if "未找到可用路径" in message else 400
        raise HTTPException(status_code=status_code, detail=message)
    return result


@app.get("/api/isochrone")
async def isochrone(
    lng: float = Query(..., description="中心点经度，4326"),
    lat: float = Query(..., description="中心点纬度，4326"),
    mode: str = Query("car", pattern="^(car|walk)$"),
    time_limit: int = Query(1800, ge=60, le=14400, description="时间阈值，单位秒"),
):
    cost_field = "cost_car_s" if mode == "car" else "cost_walk_s"

    conn = None
    cur = None
    try:
        conn = get_db_connection()
        cur = conn.cursor()

        nearest_vertex_sql = """
        SELECT id
        FROM fuzhou_roads_vertices
        ORDER BY geom <-> ST_Transform(ST_SetSRID(ST_MakePoint(%s, %s), 4326), 3857)
        LIMIT 1;
        """
        cur.execute(nearest_vertex_sql, (lng, lat))
        nearest_row = cur.fetchone()
        if not nearest_row:
            raise HTTPException(status_code=404, detail="未找到可用起始节点")

        start_vid = int(nearest_row["id"])
        edge_sql = (
            f"SELECT id, source, target, {cost_field} AS cost "
            f"FROM fuzhou_roads_clean WHERE {cost_field} >= 0"
        )

        isochrone_sql = f"""
        WITH reachable AS (
            SELECT node, cost
            FROM pgr_drivingDistance(
                '{edge_sql}',
                %s,
                %s,
                FALSE
            )
        ),
        pts AS (
            SELECT v.id, v.geom, r.cost
            FROM reachable r
            JOIN fuzhou_roads_vertices v ON v.id = r.node
        ),
        hull AS (
            SELECT
                CASE
                    WHEN COUNT(*) >= 3 THEN ST_ConcaveHull(ST_Collect(geom), 0.90, TRUE)
                    WHEN COUNT(*) = 2 THEN ST_Buffer(ST_Envelope(ST_Collect(geom)), 35)
                    WHEN COUNT(*) = 1 THEN ST_Buffer(ST_Collect(geom), 35)
                    ELSE NULL
                END AS geom,
                COUNT(*) AS node_count,
                COALESCE(MAX(cost), 0) AS max_cost_s
            FROM pts
        )
        SELECT
            ST_AsGeoJSON(ST_Transform(ST_MakeValid(geom), 4326)) AS polygon_geojson,
            node_count,
            max_cost_s
        FROM hull
        WHERE geom IS NOT NULL;
        """
        cur.execute(isochrone_sql, (start_vid, time_limit))
        row = cur.fetchone()
        if not row or not row.get("polygon_geojson"):
            raise HTTPException(status_code=404, detail="未生成有效服务区，请扩大时间阈值重试")

        polygon_geometry = json.loads(row["polygon_geojson"])
        feature = {
            "type": "Feature",
            "properties": {
                "mode": mode,
                "time_limit_s": time_limit,
                "node_count": int(row.get("node_count") or 0),
                "max_cost_s": float(row.get("max_cost_s") or 0.0),
                "start_vertex": start_vid,
            },
            "geometry": polygon_geometry,
        }

        return {
            "success": True,
            "mode": mode,
            "time_limit_s": time_limit,
            "center": {"lng": lng, "lat": lat},
            "geojson": {
                "type": "FeatureCollection",
                "features": [feature],
            },
        }
    except psycopg2.Error as exc:
        raise HTTPException(status_code=500, detail=f"服务区分析失败: {exc}")
    finally:
        if cur:
            cur.close()
        if conn:
            conn.close()

        function extractLatLng(target) {
            if (!target) {
                return null;

            if (typeof target.lat === 'number' && typeof target.lng === 'number') {
                return L.latLng(target.lat, target.lng);
            }

            if (Array.isArray(target) && target.length >= 2) {

                const lat = Number(target[1]);
                if (Number.isFinite(lat) && Number.isFinite(lng)) {

            }

            if (
                target.geometry &&
                target.geometry.type === 'Point' &&
                Array.isArray(target.geometry.coordinates)

                if (Number.isFinite(Number(lat)) && Number.isFinite(Number(lng))) {
                    return L.latLng(Number(lat), Number(lng));
                }
            }

            return null;
        }

        function sumDistanceBySegment(payload, segmentType) {
            if (!payload || !payload.geojson || !Array.isArray(payload.geojson.features)) {
                return 0;
            }
            return payload.geojson.features.reduce((acc, feature) => {
                const props = feature && feature.properties ? feature.properties : {};
                if (props.segment_type !== segmentType) {
                    return acc;
                }
                return acc + (Number(props.distance_m) || 0);
            }, 0);
        }

        async function loadIsochrone(mode, seconds) {

                throw new Error('请先设置虚拟起点，再进行服务区分析');
            }

            const params = new URLSearchParams({
                lng: String(currentLocation.lng),
                lat: String(currentLocation.lat),
                mode,
                time_limit: String(seconds)
            });
            const res = await fetch(`/api/isochrone?${params.toString()}`);
            const payload = await parseApiResponse(res, '服务区分析失败');

            const fc = payload && payload.geojson ? payload.geojson : null;
            if (!fc || fc.type !== 'FeatureCollection') {
                throw new Error('服务区返回格式错误，需为标准 FeatureCollection');
            }

            const layer = isochroneLayers[mode][seconds];
            layer.clearLayers();

            const style = getIsochroneStyle(mode, seconds);
            const geoLayer = L.geoJSON(fc, {
                style,
                onEachFeature: (feature, lyr) => {
                    const props = feature && feature.properties ? feature.properties : {};
                    const mins = Math.round((Number(props.time_limit_s) || seconds) / 60);
                    lyr.bindPopup(`${mode === 'car' ? '驾车' : '步行'}服务区 ${mins} 分钟`);
                }
            });
            layer.addLayer(geoLayer);
        }

                        return;
                    }
                    const key = `${lng.toFixed(6)}|${lat.toFixed(6)}`;
                    if (seen.has(key)) {
                        return;
                    }
                    seen.add(key);
                    footprintFeatures.push({
                        type: 'Feature',
                        properties: { kind },
                        geometry: {
                            type: 'Point',
                            coordinates: [lng, lat]
                        }
                    });
                };

                addPointFeature(coords[0], 'start');
                addPointFeature(coords[coords.length - 1], 'end');

                for (let i = 1; i < coords.length - 1; i += 1) {
                    const a = coords[i - 1];
                    const b = coords[i];
                    const c = coords[i + 1];
                    if (!a || !b || !c) {
                        continue;
                    }

                    const v1x = Number(b[0]) - Number(a[0]);
                    const v1y = Number(b[1]) - Number(a[1]);
                    const v2x = Number(c[0]) - Number(b[0]);
                    const v2y = Number(c[1]) - Number(b[1]);
                    const m1 = Math.hypot(v1x, v1y);
                    const m2 = Math.hypot(v2x, v2y);
                    if (!m1 || !m2) {
                        continue;
                    }
                    const cosValue = Math.max(-1, Math.min(1, (v1x * v2x + v1y * v2y) / (m1 * m2)));
                    const angleDeg = Math.acos(cosValue) * 180 / Math.PI;
                    if (angleDeg >= 30) {
                        addPointFeature(b, 'turn');
                    }
                }
            });

            if (footprintFeatures.length) {
                const footprintLayer = L.geoJSON(
                    { type: 'FeatureCollection', features: footprintFeatures },
                    {
                        pointToLayer: (_feature, latlng) => {
                            return L.marker(latlng, {
                                icon: L.divIcon({
                                    className: '',
                                    html: '<div class="footstep-marker">👣</div>',
                                    iconSize: [24, 24],
                                    iconAnchor: [12, 12]
                                })
                            });
                        }
                    }
                );
                walkRouteLayer.addLayer(footprintLayer);
            }
        }

        function fitRouteBounds() {
            const fg = L.featureGroup();
            if (carRouteLayer.getLayers().length) {
                fg.addLayer(carRouteLayer);
            }
            if (walkRouteLayer.getLayers().length) {
                fg.addLayer(walkRouteLayer);
            }
            if (fg.getLayers().length) {
                map.fitBounds(fg.getBounds(), { padding: [40, 40], maxZoom: 16 });
            }
        }

        async function loadCategories() {
            try {
                const res = await fetch('/api/categories');
                const payload = await parseApiResponse(res, '分类加载失败');

                const categorySet = new Set(['休闲服务', '教育科研', '医疗卫生']);
                (payload.data || []).forEach(c => categorySet.add(c));

                categorySelect.innerHTML = '<option value="">全部类型</option>';
                Array.from(categorySet)
                    .filter(Boolean)
                    .sort((a, b) => a.localeCompare(b, 'zh-CN'))
                    .forEach(category => {
                        const option = document.createElement('option');
                        option.value = category;
                        option.textContent = category;
                        categorySelect.appendChild(option);
                    });
            } catch (err) {
                showError(`无法加载分类: ${err.message}`);
            }
        }

        function renderResults(data) {
            resultLayer.clearLayers();
            poiMarkerByKey = new Map();
            const bounds = [];

            data.forEach(item => {
                const poiPoint = parsePoiPointCoordinates(item);
                if (!poiPoint) return;
                const coords = [poiPoint.lng, poiPoint.lat];
                const latlng = [coords[1], coords[0]];
                bounds.push(latlng);

                const visual = categoryStyle(item.category || '');
                const marker = L.marker(latlng, {
                    icon: L.divIcon({
                        className: 'poi-icon-wrap',
                        html: `<div class="poi-pin ${visual.key}">${visual.icon}</div>`,
                        iconSize: [18, 18],
                        iconAnchor: [9, 9]
                    })
                })
                    .addTo(resultLayer)
                    .bindPopup(
                        `<b>${item.name || '未命名地点'}</b><br>类型: ${item.category || '未知'}${typeof item.distance_m === 'number' ? `<br>距离: ${Math.round(item.distance_m)} 米` : ''}`
                        + `<br><button class="to-here-btn" onclick="to_here(${coords[1]}, ${coords[0]})">导航到这里</button>`
                    )
                    .on('click', () => {
                        destinationPoi = item;
                        if (!currentLocation) {
                            showError('请先选择虚拟地点，再点击地点进行导航。');
                            return;
                        }
                        requestRoute();
                    });

                    const key = getPoiKey(item, coords);
                    poiMarkerByKey.set(key, marker);
            });

            if (bounds.length) {
                map.fitBounds(bounds, { padding: [30, 30], maxZoom: 16 });
            }
        }

        function renderSearchResults(data) {
            searchResultsEl.innerHTML = '';
            if (!data.length) {
                searchResultsEl.classList.remove('show');
                return;
            }

            data.forEach((item, index) => {
                const li = document.createElement('li');
                li.innerHTML = `<strong>${item.name || '未命名地点'}</strong><span>${item.category || '未知类型'}</span>`;
                li.addEventListener('click', () => {
                    focusOnResult(index);
                    searchResultsEl.classList.remove('show');
                });
                searchResultsEl.appendChild(li);
            });

            searchResultsEl.classList.add('show');
        }

        function focusOnResult(index) {
            const item = currentResults[index];
            const poiPoint = parsePoiPointCoordinates(item);
            if (!item || !poiPoint) return;

            destinationPoi = item;
            const latlng = [poiPoint.lat, poiPoint.lng];
            map.setView(latlng, 16, { animate: true });

            if (currentLocation) {
                requestRoute();
            }
        }

        async function searchPlaces() {
            const keyword = searchInput.value.trim();
            if (!keyword) {
                searchResultsEl.classList.remove('show');
                return;
            }

            try {
                hideError();
                const params = new URLSearchParams({ keyword, limit: '30' });
                if (categorySelect.value) {
                    params.set('category', categorySelect.value);
                }

                const res = await fetch(`/api/search?${params.toString()}`);
                const payload = await parseApiResponse(res, '地点搜索失败');
                currentResults = payload.data || [];
                renderResults(currentResults);
                renderSearchResults(currentResults);
            } catch (err) {
                showError(`搜索失败: ${err.message}`);
            }
        }

        async function updateNearbySearch() {
            if (!currentLocation) {
                return;
            }

            const radius = Math.max(100, Math.min(20000, Number(radiusInput.value) || 2000));
            radiusInput.value = radius;
            currentRadius = radius;
            sidebarRadiusValue.textContent = String(currentRadius);

            try {
                hideError();
                const params = new URLSearchParams({
                    lng: String(currentLocation.lng),
                    lat: String(currentLocation.lat),
                    radius: String(radius),
                    limit: '200'
                });
                if (categorySelect.value) {
                    params.set('category', categorySelect.value);
                }

                const res = await fetch(`/api/nearby?${params.toString()}`);
                const payload = await parseApiResponse(res, '附近地点检索失败');
                currentResults = payload.data || [];
                renderResults(currentResults);
                renderNearbySidebar(currentResults);
            } catch (err) {
                showError(`附近地点检索失败: ${err.message}`);
            }
        }

        const debouncedNearbySearch = debounce(updateNearbySearch, 320);

        function setCurrentLocation(latlng, popupText = '已设置为虚拟当前位置') {
            currentLocation = { lat: latlng.lat, lng: latlng.lng };

            userMarker.clearLayers();
            const currentMarker = L.marker(latlng, {
                icon: L.divIcon({ className: 'user-icon', html: '📍', iconSize: [28, 28] })

                .addTo(userMarker)
                .bindPopup(popupText)
                .openPopup();

            setTimeout(() => {
                if (currentMarker && typeof currentMarker.closePopup === 'function') {
                    currentMarker.closePopup();
                }
            }, 5000);

            updateNearbySearch();
            if (destinationPoi) {
                requestRoute();
            }
        }

        function getAnalyzeCategories() {

            }
            return Array.from(categorySelect.options)
                .map(option => option.value)
                .filter(Boolean);
        }

        async function analyzeGeo(geojson, type) {
            try {
                hideError();
                const response = await fetch('/api/analyze', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        geojson,
                        type,
                        categories: getAnalyzeCategories()

            }
        }

        async function requestRoute() {
            if (!currentLocation || !destinationPoi) {
                return;
            }

            try {
                hideError();
                const poiPoint = parsePoiPointCoordinates(destinationPoi);
                if (!poiPoint) {
                    throw new Error('终点坐标无效，请重新选择 POI 点位');
                }
                const baseParams = {
                    origin_lng: String(currentLocation.lng),
                    origin_lat: String(currentLocation.lat),
                    dest_lng: String(poiPoint.lng),
                    dest_lat: String(poiPoint.lat)
                };

                const carParams = new URLSearchParams({ ...baseParams, mode: 'car' });
                const walkParams = new URLSearchParams({ ...baseParams, mode: 'walk' });

                const [carResult, walkResult] = await Promise.allSettled([
                    fetch(`/api/route_shortest?${carParams.toString()}`),
                    fetch(`/api/route_shortest?${walkParams.toString()}`)
                ]);

                let carPayload = null;
                let walkPayload = null;
                const errors = [];

                if (carResult.status === 'fulfilled') {
                    try {
                        carPayload = await parseApiResponse(carResult.value, '驾车路径规划失败');
                    } catch (err) {
                        errors.push(`驾车: ${err.message}`);
                    }
                } else {
                    errors.push(`驾车: ${carResult.reason && carResult.reason.message ? carResult.reason.message : '请求失败'}`);
                }

                if (walkResult.status === 'fulfilled') {
                    try {
                        walkPayload = await parseApiResponse(walkResult.value, '步行路径规划失败');
                    } catch (err) {
                        errors.push(`步行: ${err.message}`);
                    }
                } else {
                    errors.push(`步行: ${walkResult.reason && walkResult.reason.message ? walkResult.reason.message : '请求失败'}`);
                }

                if (!carPayload && !walkPayload) {
                    throw new Error(errors.join('；') || '路径规划失败');
                }

                drawCarRoute(carPayload);
                drawWalkRoute(walkPayload);
                applyLayerVisibility();
                updateMetrics(carPayload, walkPayload);
                fitRouteBounds();

                if (errors.length) {
                    showError(`部分模式规划失败: ${errors.join('；')}`);
                }
            } catch (err) {
                showError(`路径规划失败: ${err.message}`);
            }
        }
